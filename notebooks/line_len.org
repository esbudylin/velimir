  from collections import defaultdict

  def stem_leaf(data, stem_size=10):
      """
      stem_size=10 means:
        stem = tens digit (or groups of 10)
        leaf = units digit
      """
      stems = defaultdict(list)

      for x in sorted(data):
          stem = x // stem_size
          leaf = x % stem_size
          stems[stem].append(leaf)

      print("\nStem-and-Leaf Plot (stem=//10, leaf=%10)\n")
      for stem in sorted(stems):
          leaves = " ".join(str(leaf) for leaf in stems[stem])
          print(f"{stem:>3} | {leaves}")
