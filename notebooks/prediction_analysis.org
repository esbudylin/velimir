  SELECT
      feet,
      ROUND(AVG(meter_pred = meter_target), 3) AS correct_ratio,
      COUNT(*) as cnt
  FROM (
      SELECT
          meter_pred,
          meter_target,
          LENGTH(REPLACE(accent_target, '0', '')) AS feet

      WHERE meter_target <> 'С' -- Исключаем силлабику
