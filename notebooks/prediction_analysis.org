#+begin_src sqlite
  CREATE VIEW error_details AS
  SELECT
      p.*,
      CASE
          WHEN caesura_pred <> '' THEN 'caesura'
          WHEN meter_pred IN ('Ак', 'Тк', 'Дк') THEN 'tonic'
          WHEN meter_pred IN ('Я', 'Х', 'Аф', 'Ан', 'Д') THEN 'syllabo_tonic'
          ELSE 'other'
      END AS stratum
  FROM predictions p
  WHERE meter_class_target <> meter_class_pred;
#+end_src

#+begin_src sqlite
  CREATE VIEW error_strata AS
  SELECT
      meter_class_target,
      meter_target,
      COUNT(*) AS total_errors,
      ROUND(SUM(CASE WHEN stratum = 'caesura' THEN 1 ELSE 0 END) * 1.0 / COUNT(*), 4) AS caesura_pct,
      ROUND(SUM(CASE WHEN stratum = 'tonic' THEN 1 ELSE 0 END) * 1.0 / COUNT(*), 4) AS tonic_pct,
      ROUND(SUM(CASE WHEN stratum = 'syllabo_tonic' THEN 1 ELSE 0 END) * 1.0 / COUNT(*), 4) AS syllabo_tonic_pct,
      ROUND(SUM(CASE WHEN stratum = 'other' THEN 1 ELSE 0 END) * 1.0 / COUNT(*), 4) AS other_pct
  FROM error_details
  GROUP BY meter_class_target
  ORDER BY total_errors DESC;

#+begin_src sqlite
  SELECT
      feet,
      ROUND(AVG(meter_pred = meter_target), 3) AS correct_ratio,
      COUNT(*) as cnt
  FROM (
      SELECT
          meter_pred,
          meter_target,
          LENGTH(REPLACE(accent_target, '0', '')) AS feet
      FROM predictions
      WHERE meter_target <> 'С' -- Исключаем силлабику
