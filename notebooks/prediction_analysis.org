#+begin_src sqlite
  CREATE VIEW error_details AS
  SELECT
      p.*,
      CASE
          WHEN caesura_target <> '' THEN 'caesura'
          WHEN meter_target IN ('Ак', 'Тк', 'Дк') THEN 'tonic'
          WHEN meter_target IN ('Я', 'Х', 'Аф', 'Ан', 'Д') THEN 'syllabo_tonic'
          ELSE 'other'
      END AS target_category,
      CASE
          WHEN caesura_pred <> '' THEN 'caesura'
          WHEN meter_pred IN ('Ак', 'Тк', 'Дк') THEN 'tonic'
          WHEN meter_pred IN ('Я', 'Х', 'Аф', 'Ан', 'Д') THEN 'syllabo_tonic'
          ELSE 'other'
      END AS pred_category
  FROM predictions p
  WHERE meter_class_target <> meter_class_pred;

  SELECT
      feet,
      ROUND(AVG(meter_pred = meter_target), 3) AS correct_ratio,
      COUNT(*) as cnt

      SELECT
          meter_pred,
          meter_target,
