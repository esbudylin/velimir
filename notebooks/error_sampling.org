  from velimir.accentuator import is_vowel, stress_mark_ord, remove_accent_marks

  def put_accents(line: str, mask: list[bool]):
      res = ""
      vowel_pos = 0

      for c in remove_accent_marks(line):
          res += c
          if is_vowel(c):
              if mask[vowel_pos]:
                  res += chr(stress_mark_ord)
              vowel_pos += 1

      return res

  from bitarray import bitarray

  from velimir.parsers import extract_syllable_features, parse_input_lines
  from velimir.io import read_poem_xml
  from velimir.evaluation import meters_to_str

  samples = [("Файл", "Страта", "Метр (корпус)", "Предсказанный метр", "Текст", "Разметка ударений", "Поэтические ударения")]

  for target_category, pred_category, n_samples in allocations:
      stratum = stratum_labels[target_category] + " → " + stratum_labels[pred_category]

      result = conn.execute("""
          SELECT poem_path, line_idx, meter_class_target, meter_class_pred, accent_pred
          FROM error_details
          WHERE target_category = ? AND pred_category = ?
          ORDER BY md5(poem_path || ':' || line_idx || ':424')
          LIMIT ?
      """, (target_category, pred_category, n_samples))

      for poem_path, line_id, target_class, pred_class, accent_pred in result:
          target_mc, pred_mc = map(MeterClassRegistry.int_to_mc, (target_class, pred_class))

          poem_xml = read_poem_xml(poem_path)

          input_lines, _ = parse_input_lines(poem_xml)
          line = next(line for line in input_lines if line.idx == line_id)

          caesura_to_str = lambda c: "-".join(map(str, c.caesura)) or ''

          ling_accents = put_accents(line.text, extract_syllable_features(line.text).linguistic_accents)
          poetic_accents = put_accents(line.text, bitarray(accent_pred))

          samples.append((
              poem_path,
              stratum,
              meters_to_str(target_mc) + " " + caesura_to_str(target_mc),
              meters_to_str(pred_mc) + " " + caesura_to_str(pred_mc),
              line.text,
              ling_accents,
              poetic_accents,
          ))

  samples
