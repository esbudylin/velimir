  from velimir.accentuator import is_vowel, stress_mark_ord, remove_accent_marks

  def put_accents(line: str, mask: list[bool]):
      res = ""
      vowel_pos = 0

      for c in remove_accent_marks(line):
          res += c
          if is_vowel(c):
              if mask[vowel_pos]:
                  res += chr(stress_mark_ord)
              vowel_pos += 1

      return res

  from bitarray import bitarray

  from velimir.parsers import extract_syllable_features, parse_input_lines
  from velimir.io import read_poem_xml
  from velimir.evaluation import meters_to_str
  from velimir.ml_loader import MeterClassRegistry
  from velimir.settings import PREDICTION_DB_PATH

  TOTAL_SAMPLES = 100

  MeterClassRegistry.initialize()

  stratum_labels = {
      "syllabo_tonic":   "Силлабо-тоника",
      "tonic":           "Тоника",
      "caesura":         "Цезура",
      "other":           "Прочее",
  }

  rows = conn.execute("""
      SELECT meter_class_target, meter_target, total_errors, caesura_pct, tonic_pct, syllabo_tonic_pct, other_pct
      FROM error_strata
      ORDER BY total_errors DESC
  """).fetchall()

  total_all_errors = sum(r[2] for r in rows)

  allocations = []
  for mc_target, meter_target, n_err, c_pct, t_pct, st_pct, o_pct in rows:
      n_target = round(n_err / total_all_errors * TOTAL_SAMPLES)
      for key, pct in [("caesura", c_pct), ("tonic", t_pct), ("syllabo_tonic", st_pct), ("other", o_pct)]:
          n = round(pct * n_target)
          if n > 0:
              allocations.append((mc_target, meter_target, key, n))

  samples = [("Страта", "Метр (корпус)", "Предсказанный метр", "Цезура (корпус)", "Предсказанная цезура", "Текст", "Разметка ударений", "Поэтические ударения", "Файл")]

  for mc_target, meter_target, stratum_key, n_samples in allocations:
      result = conn.execute("""
          SELECT poem_path, line_idx, meter_class_target, meter_class_pred, accent_pred
          FROM error_details
          WHERE meter_class_target = ? AND stratum = ?
          ORDER BY md5(poem_path || ':' || line_idx || ':42')
          LIMIT ?
      """, (mc_target, stratum_key, n_samples))

      for poem_path, line_id, target_class, pred_class, accent_pred in result:
          target_mc, pred_mc = map(MeterClassRegistry.int_to_mc, (target_class, pred_class))

          poem_xml = read_poem_xml(poem_path)

          input_lines, _ = parse_input_lines(poem_xml)
          line = next(line for line in input_lines if line.idx == line_id)

          caesura_to_str = lambda c: "-".join(map(str, c.caesura)) or '-'

          ling_accents = put_accents(line.text, extract_syllable_features(line.text).linguistic_accents)
          poetic_accents = put_accents(line.text, bitarray(accent_pred))

          samples.append((
              stratum_labels[stratum_key],
              meters_to_str(target_mc),
              meters_to_str(pred_mc),
              caesura_to_str(target_mc),
              caesura_to_str(pred_mc),
              line.text,
              ling_accents,
              poetic_accents,
              poem_path,
          ))

  samples
