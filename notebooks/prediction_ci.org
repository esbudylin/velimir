#+begin_src R
  set.seed(986355)

  cb <- clusterBootstrap(
    df,
    clusters = c("uniq_chunk_id"),
    replace = c(TRUE),
    stat_fun = function(df) {
      class_acc <- sapply(meter_classes, function(k) {

          idx <- df$meter_class_target == k

          if (sum(idx) == 0) return(NA_real_)

          mean(df$meter_class_pred[idx] == df$meter_class_target[idx])
      })

      names(class_acc) <- meter_class_names

      c(
        accent_acc = mean(df$accent_pred == df$accent_target),
        meter_acc = mean(df$meter_class_pred == df$meter_class_target),
        class_acc
      )
    },
    B = 10000
  )
#+end_src

#+RESULTS:

#+begin_src R
  roundResults <- function(res) {
    lapply(res, function(col) {
      if (is.numeric(col)) signif(col, 3) else col
    })
  }
#+end_src

#+RESULTS:

#+begin_src R
  roundResults(
    confint(
      cb,
      level = 0.95,
      parm = c("accent_acc", "meter_acc")
    )
  )
