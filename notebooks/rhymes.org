#+begin_src sqlite
  WITH text_avg_coef AS (
      SELECT poem_id, AVG(rhyming_coef) as avg_coef FROM rhyme_annotations
      GROUP BY(poem_id)
  )
  SELECT COUNT(*) FROM rhyme_annotations ra
  JOIN text_avg_coef tac ON ra.poem_id = tac.poem_id
  WHERE
      rhyming_coef > 0.2 OR
      (rhyming_coef < 0.2 AND rhyming_coef > 0.1 AND tac.avg_coef > 0.45)
#+end_src

#+begin_src sqlite
  WITH text_avg_coef AS (
  SELECT poem_id, AVG(rhyming_coef) as avg_coef FROM rhyme_annotations
  GROUP BY(poem_id)
  )
  SELECT printf('%.1f-%.1f',
                 CAST(avg_coef * 10 AS INT) / 10.0,
                 (CAST(avg_coef * 10 AS INT) + 1) / 10.0)
  AS strata,
  COUNT(*) AS n,
  ROUND(100.0 * COUNT(*) / (SELECT COUNT(*) FROM text_avg_coef), 2) AS pct
  FROM text_avg_coef
  GROUP BY strata
  ORDER BY MIN(avg_coef)
#+end_src

  SELECT

      WHEN rhyming_coef = 0.0 THEN '0.0'
      WHEN rhyming_coef = 1.0 THEN '1.0'
      ELSE printf('%.1f-%.1f',
                  CAST(rhyming_coef * 10 AS INT) / 10.0,
                  (CAST(rhyming_coef * 10 AS INT) + 1) / 10.0)
    END AS strata,
    COUNT(*) AS n,
    ROUND(100.0 * COUNT(*) / (SELECT COUNT(*) FROM rhyme_annotations), 2) AS pct
  FROM rhyme_annotations
  GROUP BY strata
  ORDER BY MIN(rhyming_coef)
