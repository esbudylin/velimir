#+begin_src sqlite
  WITH text_avg_coef AS (
  SELECT poem_id, AVG(rhyming_coef) as avg_coef FROM rhyme_annotations
  GROUP BY(poem_id)
  )
  SELECT printf('%.1f-%.1f',
                 CAST(avg_coef * 10 AS INT) / 10.0,
                 (CAST(avg_coef * 10 AS INT) + 1) / 10.0)
  AS strata,
  COUNT(*) AS n,
  ROUND(100.0 * COUNT(*) / (SELECT COUNT(*) FROM text_avg_coef), 2) AS pct
  FROM text_avg_coef
  GROUP BY strata
  ORDER BY MIN(avg_coef)
#+end_src

    END AS strata,
    COUNT(*) AS n,
    ROUND(100.0 * COUNT(*) / (SELECT COUNT(*) FROM rhyme_annotations), 2) AS pct
  FROM rhyme_annotations
  GROUP BY strata

#+begin_src sqlite
  CREATE VIEW excluded_rhymes AS
  WITH text_avg_coef AS (
      SELECT poem_id, AVG(rhyming_coef) AS avg_coef
      FROM rhyme_annotations
      GROUP BY poem_id
  )
  SELECT authors.name AS author,
         poems.path AS poem_path,
         poems.header AS title,
         json_group_array(DISTINCT r.word) AS words,
         ROUND(AVG(ra.rhyming_coef), 2) AS rhyming_coef,
         ROUND(tac.avg_coef, 2)         AS avg_coef
  FROM rhymes r
  JOIN poems ON r.poem_id = poems.ROWID
  JOIN authors ON poems.author_id = authors.ROWID
  JOIN rhyme_annotations ra
      ON ra.poem_id     = r.poem_id
     AND ra.seq         = r.seq
     AND ra.rhyme_group = r.rhyme_group
  JOIN text_avg_coef tac ON tac.poem_id = r.poem_id
  WHERE NOT EXISTS (
      SELECT 1
      FROM validated_rhyme_groups vrg
      WHERE vrg.poem_id     = r.poem_id
        AND vrg.seq         = r.seq
        AND vrg.rhyme_group = r.rhyme_group
  )
  GROUP BY r.poem_id, r.seq, r.rhyme_group
#+end_src
